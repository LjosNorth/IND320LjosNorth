AI usage¶

I in my coding work utilize AI in four primary ways.

  1. Inline suggestions. these help code faster, though ofc often they are also completely wrong so its not a brain off usage thing and one must be aware of what one is accepting.

  3. how-to's. I will often have some idea of i want to do this thing but im not sure how to do it, say how to do subplots with plotly (my experience is mostly with matplotlib).

# Create subplot grid (3 rows, 2 cols)
fig, axes = plt.subplots(3, 2, figsize=(14, 12))
axes = axes.flatten()  # flatten to 1D for easier indexing

data_columns = [
    "filled degree",
    "fill_Twh",
    "filled degree last week",
    "change in filled degree",
    "capacity_Twh"
]

for idx, col in enumerate(data_columns):
    ax = axes[idx]
    data = reservoirsLog[col]

    kde = gaussian_kde(data)
    x_range = np.linspace(data.min(), data.max(), 200 if col != "change in filled degree" else 50)
    density = kde(x_range)
    density = density / density.max()

    ax.fill_between(x_range, density, alpha=0.6, color='steelblue')
    ax.plot(x_range, density, color='steelblue', linewidth=2)
    ax.set_title(col, fontsize=12, fontweight='bold')
    ax.set_ylabel('Density', fontsize=10)
    ax.set_xlabel('Value', fontsize=10)
    ax.grid(True, alpha=0.3)

# Hide the empty subplot (6th position)
axes[-1].set_visible(False)

plt.tight_layout()
plt.show()

# Create single figure
fig, ax = plt.subplots(figsize=(12, 7))

data_columns = [
    ("filled degree", 200),
    ("fill_Twh", 200),
    ("filled degree last week", 200),
    ("change in filled degree", 50),
    ("capacity_Twh", 200)
]

colors = ['steelblue', 'coral', 'green', 'purple', 'orange']

for (col, x_points), color in zip(data_columns, colors):
    data = reservoirsLog[col]

    kde = gaussian_kde(data)
    x_range = np.linspace(data.min(), data.max(), x_points)
    density = kde(x_range)
    density = density / density.max()

    ax.plot(x_range, density, label=col, linewidth=2, color=color)
    ax.fill_between(x_range, density, alpha=0.2, color=color)

ax.set_xlabel('Value', fontsize=12)
ax.set_ylabel('Density (normalized)', fontsize=12)
ax.set_title('Kernel Density Estimates - Reservoirs Data', fontsize=14, fontweight='bold')
ax.legend(loc='upper right', fontsize=10)
ax.grid(True, alpha=0.3)

plt.tight_layout()
plt.show()
